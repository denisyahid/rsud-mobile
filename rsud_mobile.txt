function generateUuid() {
    return sprintf('%08x-%04x-%04x-%04x-%04x%08x',
        mt_rand(0, 0xffffffff),
        mt_rand(0, 0xffff),
        mt_rand(0, 0xffff),
        mt_rand(0, 0xffff),
        mt_rand(0, 0xffff),
        mt_rand(0, 0xffffffff)
    );
}

function generateStrukNumber($pdo, $kdProfile = 1) {
    try {
        $stmt = $pdo->prepare("
            SELECT COALESCE(MAX(CAST(SUBSTRING(nostruk FROM 2) AS BIGINT)), 0) AS max_no 
            FROM strukpelayanan_t 
            WHERE kdprofile = :kdprofile 
              AND nostruk LIKE 'S%'
              AND SUBSTRING(nostruk FROM 2) ~ '^[0-9]+$'
        ");
        $stmt->execute([':kdprofile' => $kdProfile]);
        $result = $stmt->fetch();
        $maxNo = (int)($result['max_no'] ?? 0);
        return 'S' . str_pad($maxNo + 1, 9, '0', STR_PAD_LEFT);
    } catch (Exception $e) {
        return 'S' . date('YmdHis') . rand(100, 999);
    }
}

function getSettingFix($pdo, $key) {
    $stmt = $pdo->prepare("SELECT value FROM settingdatafixed_m WHERE key = :key AND statusenabled = true LIMIT 1");
    $stmt->execute([':key' => $key]);
    $result = $stmt->fetch();
    return $result ? $result['value'] : null;
}

function generateDailyBarcode() {
    // Format: YYYYMMDD + random 4 digit + checksum
    $date = date('Ymd');
    $random = str_pad(mt_rand(0, 9999), 4, '0', STR_PAD_LEFT);
    $base = $date . $random;
    // Simple checksum: sum of digits mod 10
    $sum = 0;
    for ($i = 0; $i < strlen($base); $i++) {
        $sum += intval($base[$i]);
    }
    $checksum = $sum % 10;
    return $base . $checksum;
}

function validateDailyBarcode($code) {
    // Check length
    if (strlen($code) !== 13) return false;
    // Check date part (first 8 digits)
    $datePart = substr($code, 0, 8);
    $today = date('Ymd');
    if ($datePart !== $today) return false;
    // Validate checksum
    $base = substr($code, 0, 12);
    $checksum = intval(substr($code, 12, 1));
    $sum = 0;
    for ($i = 0; $i < strlen($base); $i++) {
        $sum += intval($base[$i]);
    }
    return ($sum % 10) === $checksum;
}

            $j['sisa_quota'] = max(0, (int)$j['quota'] - (int)$j['terpakai']);
            $j['status'] = $j['sisa_quota'] > 0 ? 'Tersedia' : 'Penuh';
        }
        
        if ($mode === 'minggu') {
            $grouped = [];
            foreach ($jadwalList as $j) {
                $tgl = $j['tanggal'];
                if (!isset($grouped[$tgl])) {
                    $grouped[$tgl] = [
                        'tanggal' => $tgl,
                        'hari' => $j['hari'],
                        'jadwal' => [],
                    ];
                }
                $grouped[$tgl]['jadwal'][] = $j;
            }
            respond(['success' => true, 'data' => array_values($grouped)]);
        } else {
            respond(['success' => true, 'data' => $jadwalList]);
        }
        break;

    // ============================================================
    // CHECK ACTIVE BOOKING
    // ============================================================
    case 'check_active_booking':
        if (!isLoggedIn()) {
            respond(['error' => 'Unauthorized'], 401);
        }
        $pasien_id = $_SESSION['pasien_id'];
        
        $stmt = $pdo->prepare("
            SELECT COUNT(*) AS total 
            FROM pasiendaftar_t 
            WHERE nocmfk = :pasien_id 
              AND DATE(tglregistrasi) = CURRENT_DATE 
              AND statusenabled = true 
              AND tglpulang IS NULL
        ");
        $stmt->execute([':pasien_id' => $pasien_id]);
        $result = $stmt->fetch();
        
        $hasActive = ($result['total'] ?? 0) > 0;
        
        respond([
            'success' => true,
            'has_active_booking' => $hasActive,
            'message' => $hasActive
                ? 'Anda masih memiliki kunjungan aktif hari ini yang belum selesai.'
                : 'Tidak ada reservasi aktif.'
        ]);
        break;

    // ============================================================
    // DEBUG ENDPOINT
    // ============================================================
    case 'debug':
        respond([
            'success' => true,
            'session' => $_SESSION,
            'session_id' => session_id(),
            'is_logged_in' => isLoggedIn(),
            'timestamp' => date('Y-m-d H:i:s')
        ]);
        break;

    // ============================================================
    // DEFAULT
    // ============================================================
    default:
        respond(['error' => 'Aksi tidak valid: ' . $action], 400);
}
