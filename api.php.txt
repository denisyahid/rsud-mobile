function respond($data, $status = 200) {

/**
 * Generate Struk Number
 */
function generateStrukNumber($pdo, $kdProfile = 1) {
    try {
        $stmt = $pdo->prepare("
            SELECT COALESCE(
                MAX(CAST(SUBSTRING(nostruk FROM 2) AS BIGINT)), 
                0
            ) AS max_no 
            FROM strukpelayanan_t 
            WHERE kdprofile = :kdprofile 
              AND nostruk LIKE 'S%'
              AND LENGTH(nostruk) > 1
              AND SUBSTRING(nostruk FROM 2) ~ '^[0-9]+$'
        ");
        $stmt->execute([':kdprofile' => $kdProfile]);
        $result = $stmt->fetch();
        $maxNo = (int)($result['max_no'] ?? 0);
        $nextNo = $maxNo + 1;
        return 'S' . str_pad($nextNo, 9, '0', STR_PAD_LEFT);
    } catch (Exception $e) {
        error_log("Error generating struk number: " . $e->getMessage());
        // Fallback: gunakan timestamp
        return 'S' . date('YmdHis') . rand(100, 999);
    }
}

/**
 * Validate Checkin Barcode - Lebih fleksibel
 */
function validateCheckinBarcode($barcode) {
    error_log("Validating barcode: " . $barcode);
    
    // Format: CHECKIN-RSUD-MALANGBONG-YYYYMMDD
    $pattern = '/^CHECKIN-RSUD-MALANGBONG-(\d{8})$/';
    
    if (!preg_match($pattern, $barcode, $matches)) {
        error_log("Barcode format invalid: " . $barcode);
        return false;
    }
    
    $datePart = $matches[1];
    $today = date('Ymd');
    $yesterday = date('Ymd', strtotime('-1 day'));
    $tomorrow = date('Ymd', strtotime('+1 day'));
    
    // Validasi: H-1, hari ini, atau H+1 (flexible untuk perbedaan timezone)
    $isValid = in_array($datePart, [$yesterday, $today, $tomorrow]);
    
    error_log("Barcode date: $datePart, Today: $today, Valid: " . ($isValid ? 'true' : 'false'));
    
    return $isValid;
}

            SELECT 
                pd.noregistrasi,
                pd.tglregistrasi,
                r.namaruangan AS poliklinik,
                p.namalengkap AS dokter,
                ap.noantrian,
                ap.prefixnoantrian,
                CONCAT(ap.prefixnoantrian, '-', LPAD(ap.noantrian::text, 3, '0')) AS noantrian_full
            FROM pasiendaftar_t pd
            LEFT JOIN antrianpasiendiperiksa_t ap ON ap.noregistrasi = pd.noregistrasi
            LEFT JOIN ruangan_m r ON r.id = pd.objectruanganlastfk
            LEFT JOIN pegawai_m p ON p.id = pd.objectpegawaifk
            WHERE pd.noregistrasi = :noregistrasi
              AND pd.nocmfk = :pasien_id
              AND pd.statusenabled = true
            LIMIT 1
        ");
        $stmt->execute([':noregistrasi' => $noregistrasi, ':pasien_id' => $pasien_id]);
        $data = $stmt->fetch();
        if (!$data) {
            respond(['error' => 'Data tidak ditemukan']);
        }
        respond(['success' => true, 'data' => $data]);
        break;

    // ---------- BATAL RESERVASI KUNJUNGAN ----------
    case 'cancel_reservation':
        if (!isLoggedIn()) {
            respond(['error' => 'Unauthorized'], 401);
        }
        if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
            respond(['error' => 'Method not allowed'], 405);
        }
        $input = json_decode(file_get_contents('php://input'), true);
        $noregistrasi = trim($input['noregistrasi'] ?? '');
        if (empty($noregistrasi)) {
            respond(['error' => 'noregistrasi required']);
        }
        $pasien_id = $_SESSION['pasien_id'];
        try {
            $pdo->beginTransaction();
            $stmt = $pdo->prepare("
                SELECT tglregistrasi, statusenabled, objectpegawaifk, objectruanganlastfk 
                FROM pasiendaftar_t 
                WHERE noregistrasi = :noregistrasi AND nocmfk = :pasien_id
            ");
            $stmt->execute([':noregistrasi' => $noregistrasi, ':pasien_id' => $pasien_id]);
            $reg = $stmt->fetch();
            if (!$reg) {
                $pdo->rollBack();
                respond(['error' => 'Data tidak ditemukan']);
            }
            if (!$reg['statusenabled']) {
                $pdo->rollBack();
                respond(['error' => 'Reservasi sudah dibatalkan atau tidak aktif']);
            }
            $stmtApd = $pdo->prepare("
                UPDATE antrianpasiendiperiksa_t 
                SET statusantrian = '3' 
                WHERE noregistrasi = :noregistrasi
            ");
            $stmtApd->execute([':noregistrasi' => $noregistrasi]);
            $stmtReg = $pdo->prepare("
                UPDATE pasiendaftar_t 
                SET statusenabled = false,
                    tglpulang = NOW()
                WHERE noregistrasi = :noregistrasi
            ");
            $stmtReg->execute([':noregistrasi' => $noregistrasi]);
            $dokterId = $reg['objectpegawaifk'];
            $ruanganId = $reg['objectruanganlastfk'];
            $tglRegistrasi = date('Y-m-d', strtotime($reg['tglregistrasi']));
            if ($dokterId && $ruanganId && $tglRegistrasi) {
                $stmtQuota = $pdo->prepare("
                    UPDATE jadwaldokter_m 
                    SET quota = quota + 1 
                    WHERE tanggal = :tanggal 
                      AND objectpegawaifk = :dokter 
                      AND objectruanganfk = :ruangan 
                      AND statusenabled = true 
                      AND kdprofile = 1
                ");
                $stmtQuota->execute([
                    ':tanggal' => $tglRegistrasi,
                    ':dokter' => $dokterId,
                    ':ruangan' => $ruanganId
                ]);
            }
            $pdo->commit();
            respond(['success' => true, 'message' => 'Reservasi berhasil dibatalkan']);
        } catch (Exception $e) {
            if ($pdo->inTransaction()) $pdo->rollBack();
            respond(['error' => 'Gagal membatalkan: ' . $e->getMessage()], 500);
        }
        break;

    // ---------- CEK JADWAL DOKTER ----------
    case 'cek_jadwal_dokter':
        $tanggal = trim($_GET['tanggal'] ?? '');
        $ruangan_id = trim($_GET['ruangan_id'] ?? '');
        $dokter_id = trim($_GET['dokter_id'] ?? '');

        if (empty($tanggal)) {
            respond(['exists' => false, 'message' => 'Tanggal wajib diisi']);
        }
        if (empty($ruangan_id)) {
            respond(['exists' => false, 'message' => 'Ruangan wajib dipilih']);
        }
        if (empty($dokter_id)) {
            respond(['exists' => false, 'message' => 'Dokter wajib dipilih']);
        }

        $stmt = $pdo->prepare("SELECT id, jammulai, jamakhir, hari, quota FROM jadwaldokter_m 
                               WHERE tanggal = :tanggal 
                               AND objectruanganfk = :ruangan 
                               AND objectpegawaifk = :dokter 
                               AND statusenabled = true 
                               AND kdprofile = 1 
                               LIMIT 1");
        $stmt->execute([
            ':tanggal' => $tanggal,
            ':ruangan' => $ruangan_id,
            ':dokter' => $dokter_id
        ]);
        $jadwal = $stmt->fetch();

        if ($jadwal) {
            respond([
                'exists' => true,
                'message' => 'Jadwal tersedia: ' . $jadwal['jammulai'] . ' - ' . $jadwal['jamakhir'] . ' (' . $jadwal['hari'] . ')',
                'data' => $jadwal
            ]);
        } else {
            respond([
                'exists' => false,
                'message' => 'Tidak ada jadwal praktik dokter untuk tanggal, ruangan, dan dokter yang dipilih.'
            ]);
        }
        break;

    // ---------- CEK BOOKING AKTIF ----------
    case 'check_active_booking':
        if (!isLoggedIn()) {
            respond(['error' => 'Unauthorized'], 401);
        }
        $pasien_id = $_SESSION['pasien_id'];
        
        $stmt = $pdo->prepare("
            SELECT COUNT(*) AS total 
            FROM pasiendaftar_t 
            WHERE nocmfk = :pasien_id 
              AND DATE(tglregistrasi) = CURRENT_DATE 
              AND statusenabled = true 
              AND tglpulang IS NULL
        ");
        $stmt->execute([':pasien_id' => $pasien_id]);
        $result = $stmt->fetch();
        
        $hasActive = ($result['total'] ?? 0) > 0;
        
        respond([
            'success' => true,
            'has_active_booking' => $hasActive,
            'message' => $hasActive
                ? 'Anda masih memiliki kunjungan aktif hari ini yang belum selesai (belum pulang). Anda tidak dapat membuat reservasi baru ke poliklinik mana pun sampai kunjungan selesai.'
                : 'Tidak ada reservasi aktif.'
        ]);
        break;

    // ---------- GET DOKTER BY JADWAL ----------
    case 'get_dokter_by_jadwal':
        $tanggal = trim($_GET['tanggal'] ?? '');
        $ruangan_id = trim($_GET['ruangan_id'] ?? '');

        if (empty($tanggal) || empty($ruangan_id)) {
            respond(['success' => false, 'message' => 'Tanggal dan ruangan wajib diisi', 'data' => []]);
        }

        $stmt = $pdo->prepare("
            SELECT DISTINCT
                jd.objectpegawaifk AS id,
                pg.namalengkap,
                jd.jammulai,
                jd.jamakhir,
                jd.hari,
                jd.quota,
                (
                    SELECT COUNT(*) 
                    FROM antrianpasiendiperiksa_t apd 
                    WHERE apd.objectpegawaifk = jd.objectpegawaifk 
                      AND apd.objectruanganfk = jd.objectruanganfk
                      AND DATE(apd.tglregistrasi) = :tanggal2
                      AND apd.statusenabled = true
                ) AS terpakai
            FROM jadwaldokter_m jd
            JOIN pegawai_m pg ON pg.id = jd.objectpegawaifk
            WHERE jd.tanggal = :tanggal
              AND jd.objectruanganfk = :ruangan
              AND jd.statusenabled = true
              AND pg.statusenabled = true
              AND jd.kdprofile = 1
            ORDER BY pg.namalengkap
        ");
        $stmt->execute([
            ':tanggal'  => $tanggal,
            ':tanggal2' => $tanggal,
            ':ruangan'  => $ruangan_id,
        ]);
        $dokterList = $stmt->fetchAll();

        foreach ($dokterList as &$d) {
            $d['sisa_quota'] = max(0, (int)$d['quota'] - (int)$d['terpakai']);
            $d['info'] = $d['jammulai'] . ' - ' . $d['jamakhir'] . 
                         ' | Sisa Kuota: ' . $d['sisa_quota'] . '/' . $d['quota'];
        }
        unset($d);

        respond(['success' => true, 'data' => $dokterList]);
        break;

    // ---------- GET OBAT (Riwayat Resep / Farmasi) ----------
    case 'get_obat':
        if (!isLoggedIn()) {
            respond(['error' => 'Unauthorized'], 401);
        }
        $pasien_id = $_SESSION['pasien_id'];

        $sqlReg = "SELECT norec, noregistrasi, tglregistrasi FROM pasiendaftar_t WHERE nocmfk = :pasien_id AND statusenabled = true";
        $stmtReg = $pdo->prepare($sqlReg);
        $stmtReg->execute([':pasien_id' => $pasien_id]);
        $registrasi = $stmtReg->fetchAll();
        $norecList = array_column($registrasi, 'norec');

        $obatList = [];
        if (!empty($norecList)) {
            $inPlaceholders = implode(',', array_fill(0, count($norecList), '?'));
            $sqlObat = "
                SELECT 
                    so.norec AS norec_so,
                    so.noorder,
                    so.tglorder,
                    so.noregistrasi,
                    so.statusorder,
                    pr.namaproduk,
                    pr.id AS produk_id,
                    op.qtyproduk AS qty,
                    op.hargasatuan,
                    djp.reportdisplay AS jenis_obat
                FROM strukorder_t so
                JOIN orderpelayanan_t op ON op.strukorderfk = so.norec AND op.statusenabled = true
                JOIN produk_m pr ON pr.id = op.objectprodukfk
                LEFT JOIN detailjenisproduk_m djp ON djp.id = pr.objectdetailjenisprodukfk
                WHERE so.noregistrasifk IN ($inPlaceholders)
                  AND so.keteranganorder = 'Order Farmasi'
                  AND so.statusenabled = true
                ORDER BY so.tglorder DESC, so.noorder
            ";
            $stmtObat = $pdo->prepare($sqlObat);
            $stmtObat->execute($norecList);
            $rawObat = $stmtObat->fetchAll();

            foreach ($rawObat as $row) {
                $key = $row['norec_so'];
                if (!isset($obatList[$key])) {
                    $obatList[$key] = [
                        'norec_so' => $row['norec_so'],
                        'noorder' => $row['noorder'],
                        'tglorder' => $row['tglorder'],
                        'noregistrasi' => $row['noregistrasi'],
                        'statusorder' => $row['statusorder'],
                        'produk' => [],
                    ];
                }
                if (!empty($row['namaproduk'])) {
                    $exists = false;
                    foreach ($obatList[$key]['produk'] as $p) {
                        if ($p['produk_id'] == $row['produk_id']) {
                            $exists = true;
                            break;
                        }
                    }
                    if (!$exists) {
                        $obatList[$key]['produk'][] = [
                            'namaproduk' => $row['namaproduk'],
                            'produk_id' => $row['produk_id'],
                            'qty' => $row['qty'] ?? 1,
                            'jenis_obat' => $row['jenis_obat'] ?? 'Obat',
                        ];
                    }
                }
            }
        }

        respond(['success' => true, 'data' => array_values($obatList)]);
        break;

    // ---------- GET JADWAL DOKTER (Public) ----------
    case 'get_jadwal_dokter':
        $tanggal     = trim($_GET['tanggal'] ?? date('Y-m-d'));
        $ruangan_id  = trim($_GET['ruangan_id'] ?? '');
        $mode        = trim($_GET['mode'] ?? 'hari');

        if ($mode === 'minggu') {
            $tanggalAkhir = date('Y-m-d', strtotime($tanggal . ' +6 days'));
            $sql = "
                SELECT
                    jd.id,
                    jd.tanggal,
                    jd.jammulai,
                    jd.jamakhir,
                    jd.hari,
                    jd.quota,
                    pg.id AS dokter_id,
                    pg.namalengkap AS namadokter,
                    ru.id AS ruangan_id,
                    ru.namaruangan,
                    ru.prefixnoantrian,
                    (
                        SELECT COUNT(*)
                        FROM antrianpasiendiperiksa_t apd
                        WHERE apd.objectpegawaifk = jd.objectpegawaifk
                          AND apd.objectruanganfk = jd.objectruanganfk
                          AND DATE(apd.tglregistrasi) = jd.tanggal
                          AND apd.statusenabled = true
                    ) AS terpakai
                FROM jadwaldokter_m jd
                JOIN pegawai_m pg ON pg.id = jd.objectpegawaifk
                JOIN ruangan_m ru ON ru.id = jd.objectruanganfk
                WHERE jd.tanggal >= :tanggal
                  AND jd.tanggal <= :tanggal_akhir
                  AND jd.tanggal >= CURRENT_DATE
                  AND jd.statusenabled = true
                  AND pg.statusenabled = true
                  AND jd.kdprofile = 1
            ";
            $params = [':tanggal' => $tanggal, ':tanggal_akhir' => $tanggalAkhir];
            if (!empty($ruangan_id)) {
                $sql .= " AND jd.objectruanganfk = :ruangan";
                $params[':ruangan'] = $ruangan_id;
            }
            $sql .= " ORDER BY jd.tanggal, ru.namaruangan, jd.jammulai";
            $stmt = $pdo->prepare($sql);
            $stmt->execute($params);
        } else {
            $sql = "
                SELECT
                    jd.id,
                    jd.tanggal,
                    jd.jammulai,
                    jd.jamakhir,
                    jd.hari,
                    jd.quota,
                    pg.id AS dokter_id,
                    pg.namalengkap AS namadokter,
                    ru.id AS ruangan_id,
                    ru.namaruangan,
                    ru.prefixnoantrian,
                    (
                        SELECT COUNT(*)
                        FROM antrianpasiendiperiksa_t apd
                        WHERE apd.objectpegawaifk = jd.objectpegawaifk
                          AND apd.objectruanganfk = jd.objectruanganfk
                          AND DATE(apd.tglregistrasi) = :tanggal2
                          AND apd.statusenabled = true
                    ) AS terpakai
                FROM jadwaldokter_m jd
                JOIN pegawai_m pg ON pg.id = jd.objectpegawaifk
                JOIN ruangan_m ru ON ru.id = jd.objectruanganfk
                WHERE jd.tanggal = :tanggal
                  AND jd.tanggal >= CURRENT_DATE
                  AND jd.statusenabled = true
                  AND pg.statusenabled = true
                  AND jd.kdprofile = 1
            ";
            $params = [':tanggal' => $tanggal, ':tanggal2' => $tanggal];
            if (!empty($ruangan_id)) {
                $sql .= " AND jd.objectruanganfk = :ruangan";
                $params[':ruangan'] = $ruangan_id;
            }
            $sql .= " ORDER BY ru.namaruangan, jd.jammulai, pg.namalengkap";
            $stmt = $pdo->prepare($sql);
            $stmt->execute($params);
        }

        $jadwalList = $stmt->fetchAll();

        foreach ($jadwalList as &$j) {
            $j['sisa_quota'] = max(0, (int)$j['quota'] - (int)$j['terpakai']);
            $j['persen_terpakai'] = $j['quota'] > 0 ? min(100, round(((int)$j['terpakai'] / (int)$j['quota']) * 100)) : 0;
            $j['status'] = $j['sisa_quota'] > 0 ? 'Tersedia' : 'Penuh';
        }
        unset($j);

        if ($mode === 'minggu') {
            $grouped = [];
            foreach ($jadwalList as $j) {
                $tgl = $j['tanggal'];
                if (!isset($grouped[$tgl])) {
                    $grouped[$tgl] = [
                        'tanggal' => $tgl,
                        'hari'    => $j['hari'],
                        'jadwal'  => [],
                    ];
                }
                $grouped[$tgl]['jadwal'][] = $j;
            }
            respond(['success' => true, 'data' => array_values($grouped)]);
        } else {
            respond(['success' => true, 'data' => $jadwalList]);
        }
        break;

    // ---------- GET POLIKLINIK LIST ----------
    case 'get_poliklinik_list':
        $poliklinik = $pdo->query("SELECT id, namaruangan, prefixnoantrian FROM ruangan_m WHERE objectdepartemenfk = 18 AND statusenabled = true ORDER BY namaruangan")->fetchAll();
        respond(['success' => true, 'data' => $poliklinik]);
        break;

    // ---------- DEBUG ENDPOINT ----------
    case 'debug':
        respond([
            'success' => true,
            'session' => $_SESSION,
            'session_id' => session_id(),
            'is_logged_in' => isLoggedIn(),
            'server' => [
                'method' => $_SERVER['REQUEST_METHOD'],
                'https' => isset($_SERVER['HTTPS']) ? $_SERVER['HTTPS'] : 'off',
                'host' => $_SERVER['HTTP_HOST'] ?? 'unknown',
                'origin' => $_SERVER['HTTP_ORIGIN'] ?? 'none'
            ],
            'barcode_test' => [
                'today' => 'CHECKIN-RSUD-MALANGBONG-' . date('Ymd'),
                'yesterday' => 'CHECKIN-RSUD-MALANGBONG-' . date('Ymd', strtotime('-1 day')),
                'validate_today' => validateCheckinBarcode('CHECKIN-RSUD-MALANGBONG-' . date('Ymd')),
                'validate_yesterday' => validateCheckinBarcode('CHECKIN-RSUD-MALANGBONG-' . date('Ymd', strtotime('-1 day'))),
            ]
        ]);
        break;

    // ---------- DEFAULT ----------
    default:
        error_log("Unknown action: " . $action);
        respond(['error' => 'Aksi tidak valid: ' . $action], 400);
}
